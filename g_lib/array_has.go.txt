
package main

import (
	"fmt"
)

func min[T comparable](a, b T) T {
	if a < b {
		return a
	}
	return b
}

func array_has[T comparable](haystack []T, needle T) bool {
	for _, val := range haystack {
		if val == needle {
			return true
		}
	}
	return false
}


func main() {
	arr := []string{"A","B","C"}
	fmt.Println(array_has(arr, "T"))
}
